一、条件语句

1. if ... else if ... else —— 成绩等级判断

if (score >= 90)      { grade = 'A（优秀）'; }
else if (score >= 80) { grade = 'B（良好）'; }
else if (score >= 70) { grade = 'C（中等）'; }
else if (score >= 60) { grade = 'D（及格）'; }
else                  { grade = 'E（不及格）'; }
结果会显示在这里

2. switch —— 今天是星期几

switch (new Date().getDay()) {
  case 0:  text = '星期日'; break;
  case 1:  text = '星期一'; break;
  case 2:  text = '星期二'; break;
  case 3:  text = '星期三'; break;
  case 4:  text = '星期四'; break;
  case 5:  text = '星期五'; break;
  case 6:  text = '星期六'; break;
  default: text = '未知';
}

二、循环语句

3. for 循环（嵌套）—— 九九乘法表

for (var i = 1; i <= 9; i++) {
  for (var j = 1; j <= i; j++) {
    // 输出 j × i = i*j
  }
}

4. while 循环 —— 从 1 累加，和不超过 1000

var sum = 0, i = 1;
while (sum + i <= 1000) {
  sum += i;
  i++;
}

5. do ... while + break / continue —— 挑选数字

do {
  n++;
  if (n % 3 === 0) { continue; } // 跳过 3 的倍数
  if (n > 16)      { break; }    // 超过 16 停止
  picked.push(n);
} while (n <= 20);
